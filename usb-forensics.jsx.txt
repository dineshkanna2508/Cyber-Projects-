function TerminalLine({ text, delay = 0, onDone }) {
  const [visible, setVisible] = useState(false);
  useEffect(() => {
    const t = setTimeout(() => { setVisible(true); if (onDone) setTimeout(onDone, 80); }, delay);
    return () => clearTimeout(t);
  }, []);
  if (!visible) return null;
  return <div className="font-mono text-sm text-green-300 leading-relaxed">{text}</div>;
}

function TypedBlock({ lines, onComplete }) {
  const [shown, setShown] = useState(0);
  const next = () => setShown(s => { const n = s + 1; if (n >= lines.length && onComplete) onComplete(); return n; });
  return (
    <div className="space-y-0.5">
      {lines.slice(0, shown + 1).map((l, i) => (
        <TerminalLine key={i} text={l} delay={i === 0 ? 100 : 0} onDone={i < lines.length - 1 ? next : (onComplete || undefined)} />
      ))}
    </div>
  );
}

        <div style={{ width: 24, height: 24, borderRadius: "50%", background: done ? "#238636" : "#1f6feb", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 11, fontWeight: 700, color: "#fff", flexShrink: 0 }}>
          {done ? "✓" : step}
        </div>
        <div>
          <div style={{ fontSize: 13, fontWeight: 600, color: "#e6edf3" }}>{title}</div>
          <div style={{ fontSize: 11, color: "#3fb950", marginTop: 2 }}>$ {cmd}</div>
        </div>
      </div>
      {children}
    </div>
  );
}

function RunBtn({ onClick, label }) {
  return (
    <button onClick={onClick} style={{ ...btnStyle("#1f6feb", "#58a6ff"), marginTop: 4 }}>{label}</button>
  );
}

    fontSize: 13, fontFamily: "inherit", cursor: "pointer", fontWeight: 600, letterSpacing: 0.5,
  };
}
